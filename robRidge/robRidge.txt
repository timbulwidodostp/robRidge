# Robust Ridge Regression Use robRidge (raiseR) With (In) R Software
install.packages("raiseR")

# Estimate Robust Ridge Regression Use robRidge (raiseR) With (In) R Software
robRidge = read.csv("https://raw.githubusercontent.com/timbulwidodostp/robRidge/main/robRidge/robRidge.csv",sep = ";")

summary(robRidge)
